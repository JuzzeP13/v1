┌─────────────────────────────────────────────────────────────────────────────┐
│ СХЕМА ТАБЛИЦЫ social_results в БД                                           │
└─────────────────────────────────────────────────────────────────────────────┘

  Колон  │ Тип      │ Назначение
  ────────┼──────────┼────────────────────────────────────────────────
  id      │ INTEGER  │ Primary Key (автоинкремент)
  url     │ TEXT     │ Ссылка на профиль (YouTube/Instagram/X)
  platform│ TEXT     │ Платформа: 'youtube', 'instagram', 'twitter'
  title   │ TEXT     │ Название профила/канала
  description TEXT  │ Описание из поиска DuckDuckGo
  username│ TEXT     │ @username или channel_id
  city    │ TEXT     │ Город поиска (Москва, СПб, etc)
  query   │ TEXT     │ Поисковый запрос ("web design", etc)
  found_at│ TEXT     │ ISO дата когда найден
  analyzed_at TEXT  │ ISO дата когда проанализирован AI
  ────────┼──────────┼────────────────────────────────────────────────
  design_score INTEGER AI оценка дизайна профила (0-10) ⭐ НОВОЕ
  ux_score    INTEGER AI оценка UX профила (0-10) ⭐ НОВОЕ
  design_text TEXT    Текстовое описание дизайна ⭐ НОВОЕ
  ux_text     TEXT    Текстовое описание UX ⭐ НОВОЕ


┌─────────────────────────────────────────────────────────────────────────────┐
│ ПРОЦЕСС СОХРАНЕНИЯ: КТО ДЕЛАЕТ ЧТО?                                        │
└─────────────────────────────────────────────────────────────────────────────┘

  social_search.py:
    search_social_media()
        │
        ├─ search_youtube_channels()     → [Найденные каналы]
        ├─ search_instagram_profiles()   → [Найденные профили]
        └─ search_twitter_accounts()     → [Найденные аккаунты]
        
                  ↓
                  
  server.py:
    process_social_search()
        │
        ├─ take_screenshot(url)          → Скриншот профила (Playwright)
        ├─ analyze_social_profile()      → AI анализ (Ollama vision)
        │                                   ├─ design_score
        │                                   ├─ ux_score
        │                                   ├─ design_text
        │                                   └─ ux_text
        ├─ INSERT INTO social_results    → Сохранение в БД
        ├─ save_social_to_excel()        → Экспорт в Excel
        └─ socketio.emit("result", {...})→ Отправка в браузер


┌─────────────────────────────────────────────────────────────────────────────┐
│ ПРИМЕРЫ ЗАПРОСОВ К БД                                                        │
└─────────────────────────────────────────────────────────────────────────────┘

  # Все результаты
  SELECT * FROM social_results;

  # Только YouTube каналы
  SELECT url, design_score, ux_score FROM social_results 
  WHERE platform='youtube' 
  ORDER BY design_score DESC;

  # Результаты из Москвы
  SELECT * FROM social_results 
  WHERE city='Москва' 
  ORDER BY analyzed_at DESC;

  # Лучшие профили (дизайн >= 7)
  SELECT url, platform, design_score, ux_score FROM social_results 
  WHERE design_score >= 7 
  ORDER BY design_score DESC;

  # Количество результатов по платформам
  SELECT platform, COUNT(*) FROM social_results 
  GROUP BY platform;

    import sqlite3
    
    conn = sqlite3.connect('tish_data.db')
    results = conn.execute(
        "SELECT url, design_score, ux_score FROM social_results "
        "WHERE platform='youtube' ORDER BY design_score DESC LIMIT 10"
    ).fetchall()

    conn.close()

┌─────────────────────────────────────────────────────────────────────────────┐
│ ЕСЛИ ЧТО-ТО НЕ РАБОТАЕТ?                                                    │
└─────────────────────────────────────────────────────────────────────────────┘

  ❌ Результаты не сохраняются в БД?
     → Проверьте что tish_data.db доступен для записи
     → Запустите: python migrate_social_results.py
     
  ❌ Excel файл не создаётся?
     → Убедитесь что директория reports/ существует
     → Проверьте права доступа на запись
     
  ❌ AI анализ не работает?
     → Ollama должна быть запущена на http://localhost:11434
     → Проверьте: curl http://localhost:11434/api/tags
     → Убедитесь что модель установлена: ollama pull qwen3-vl
     
  ❌ Скриншоты не берутся?
     → Playwright должен быть установлен (в requirements.txt)
     → Проверьте консоль Flask на ошибки [Screenshot Error]
